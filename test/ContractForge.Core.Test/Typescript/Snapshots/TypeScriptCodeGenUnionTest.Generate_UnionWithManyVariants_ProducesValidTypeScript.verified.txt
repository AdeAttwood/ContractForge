type Primitive = string | number | boolean | bigint | symbol | null | undefined;

export function flatten<T extends Record<string, unknown>>(
  obj: T,
  parentKey = "",
): Record<string, Primitive> {
  const result: Record<string, Primitive> = {};

  for (const key in obj) {
    if (!Object.hasOwn(obj, key)) continue;

    const value = obj[key];
    const newKey = parentKey ? `${parentKey}.${key}` : key;

    if (
      typeof value === "object" &&
      value !== null &&
      !Array.isArray(value)
    ) {
      Object.assign(result, flatten(value as Record<string, unknown>, newKey));
    } else {
      result[newKey] = value as Primitive;
    }
  }

  return result;
}

export interface ValidationError {
  message: string;
}
export interface NetworkError {
  code: number;
  message: string;
}
export type OperationResult =
  | { type: "success", success: string }
  | { type: "validationError", validationError: ValidationError }
  | { type: "networkError", networkError: NetworkError }
  | { type: "timeout", timeout: string }

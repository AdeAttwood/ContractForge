namespace MyApp.Domain
{
    using Microsoft.AspNetCore.Mvc;
    using System.Text.Json.Serialization;
    using System.Collections;
    using System.Threading;
    using System.Threading.Tasks;
    using System.ComponentModel.DataAnnotations;
    using Microsoft.AspNetCore.Authorization;
    public class Address
    {
        [Required]
        [JsonPropertyName("street")]
        public required string Street { get; set; }
        [Required]
        [JsonPropertyName("city")]
        public required string City { get; set; }
        [Required]
        [JsonPropertyName("state")]
        public required string State { get; set; }
        [Required]
        [JsonPropertyName("zipCode")]
        public required string ZipCode { get; set; }
    }
    public class User
    {
        [Required]
        [JsonPropertyName("name")]
        public required string Name { get; set; }
        [Required]
        [JsonPropertyName("age")]
        public required int Age { get; set; }
        [JsonPropertyName("email")]
        public  string? Email { get; set; }
        [Required]
        [JsonPropertyName("address")]
        public required Address Address { get; set; }
        [Required]
        [JsonPropertyName("phoneNumbers")]
        public required IEnumerable<string> PhoneNumbers { get; set; }
        [Required]
        [JsonPropertyName("metadata")]
        public required Dictionary<string, string> Metadata { get; set; }
    }
    public class ValidationError : Exception
    {
        [Required]
        [JsonPropertyName("code")]
        public int Code { get; set; }
        public ValidationError(int code, string message): base(message)
        {
            Code = code;
        }
    }
    [JsonConverter(typeof(JsonStringEnumConverter))]
    public enum ResultType
    {
        [JsonStringEnumMemberName("success")]
        Success,
        [JsonStringEnumMemberName("error")]
        Error,
    }
    public class Result
    {
        [JsonPropertyName("type")]
        public ResultType Type { get; set; }

        [JsonPropertyName("success")]
        [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingNull)]
        public User? Success { get; set; }
        public Result(User success)
        {
            Type = ResultType.Success;
            this.Success = success;
        }
        [JsonPropertyName("error")]
        [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingNull)]
        public ValidationError? Error { get; set; }
        public Result(ValidationError error)
        {
            Type = ResultType.Error;
            this.Error = error;
        }
    }
    public interface IUserServiceService
    {
        Task<Result> CreateUser(User user, CancellationToken cancellationToken = default);
        Task<Result> GetUser(string id, CancellationToken cancellationToken = default);
    }
    [Route("/rpc/user-service-service")]
    public abstract class UserServiceBaseController : Controller
    {
        private readonly IUserServiceService _service;
        public UserServiceBaseController(IUserServiceService service)
        {
            _service = service;
        }
        [HttpPost]
        [Route("create-user")]
        public virtual Task<Result> CreateUser([FromBody] User user, CancellationToken cancellationToken = default)
        {
            return _service.CreateUser(user, cancellationToken);
        }
        [HttpGet]
        [Route("get-user")]
        public virtual Task<Result> GetUser([FromQuery] string id, CancellationToken cancellationToken = default)
        {
            return _service.GetUser(id, cancellationToken);
        }
    }
}
